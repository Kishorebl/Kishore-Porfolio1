    <script>
        function showSection(event, sectionId) {
      document.querySelectorAll('.design-section').forEach(section => {
        section.classList.remove('active');
      });
      document.getElementById(sectionId).classList.add('active');

      document.querySelectorAll('.category-btn').forEach(btn => {
        btn.classList.remove('active');
      });
      event.currentTarget.classList.add('active');

      const headerMap = {
        'brand-logo': {
          title: 'Brand Logo Design Portfolio',
          subtitle: 'Professional logo designs showcasing brand identity and visual communication excellence'
        },
        'desktop-ui': {
          title: 'Desktop UI Design Portfolio',
          subtitle: 'User interface layouts designed for desktop platforms and software.'
        },
        'event-posters': {
          title: 'Event Posters Portfolio',
          subtitle: 'Creative and impactful posters for events, promotions, and campaigns.'
        },
        'web-interface': {
          title: 'Web Interface Design Portfolio',
          subtitle: 'Website layout and structure designs including navigation flows.'
        },
        'icon-set': {
          title: 'Icon Set Design Portfolio',
          subtitle: 'Custom icon sets designed for clarity and functionality.'
        }
      };

      const pageTitle = document.getElementById('pageTitle');
      const pageSubtitle = document.getElementById('pageSubtitle');

      if (headerMap[sectionId]) {
        pageTitle.textContent = headerMap[sectionId].title;
        pageSubtitle.textContent = headerMap[sectionId].subtitle;
      }
    }
    </script>
